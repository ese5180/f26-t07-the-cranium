cmake_minimum_required(VERSION 3.20.0)
if(NOT DEFINED BOARD)
  set(BOARD nrf7002dk/nrf5340/cpuapp CACHE STRING "CRANIUM development board")
endif()
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(cranium_telemetry)

# Work with both direct west builds and the space-free workbench wrapper.
zephyr_get(CRANIUM_PROFILE SYSBUILD)
if(NOT CRANIUM_PROFILE)
  set(CRANIUM_PROFILE optimized)
endif()
if(NOT CRANIUM_PROFILE MATCHES "^(baseline|optimized)$")
  message(FATAL_ERROR "CRANIUM_PROFILE must be baseline or optimized")
endif()
set(profile_file "${CMAKE_CURRENT_SOURCE_DIR}/profiles/${CRANIUM_PROFILE}.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${profile_file}")
file(READ "${profile_file}" profile_json)
string(JSON advertising_ms GET "${profile_json}" advertising_interval_ms)
string(JSON CRANIUM_HEARTBEAT_MS GET "${profile_json}" heartbeat_ms)
string(JSON status_led GET "${profile_json}" status_led)
if(advertising_ms LESS 20 OR advertising_ms GREATER 3200 OR
   CRANIUM_HEARTBEAT_MS LESS 0 OR CRANIUM_HEARTBEAT_MS GREATER 5000)
  message(FATAL_ERROR "Profile outside bounded BLE/heartbeat settings")
endif()
math(EXPR CRANIUM_ADV_UNITS "(${advertising_ms} * 8 + 2) / 5")
if(status_led)
  set(CRANIUM_STATUS_LED 1)
else()
  set(CRANIUM_STATUS_LED 0)
endif()
configure_file(src/profile.h.in generated/profile.h @ONLY)
target_sources(app PRIVATE src/main.c src/protocol.c)
target_include_directories(app PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
